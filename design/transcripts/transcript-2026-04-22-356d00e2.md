# ENGINE Platform - Complete Implementation Plan

## Architectural Decisions

### Framework: Chi (not Gin/Echo/Fiber)

Chi is the right choice here for four reasons specific to this use case:
1. ActivityPub requires exact HTTP header control and custom middleware chains - Chi's middleware stack is composable without framework lock-in
2. ENGINE's federation layer will need to mount sub-routers for `/.well-known/`, `/users/{actor}`, `/inbox`, `/outbox` - Chi's nested router model maps directly to this
3. Chi uses `net/http` directly, so the standard `http.Handler` interface means every ActivityPub library and crypto library works without adapters
4. No reflection magic - straightforward debugging when you're building a security-critical system

### ORM: sqlc (not GORM)

For a privacy-first system where you need to guarantee exactly what SQL runs (no hidden N+1 queries loading things you don't want loaded), sqlc is correct. You write SQL, sqlc generates type-safe Go. For the ephemeral transfer table this is essential - you need to prove the DELETE runs with certainty, not rely on GORM's soft-delete defaults being turned off correctly.

```sql
DECLARE deleted_count INTEGER;
BEGIN

    RETURN deleted_count;
```

### `004_transfers.sql` - Ephemeral Transfer Table

```sql
-- Migration: 004_transfers
-- CRITICAL PRIVACY TABLE: Records exist only during an in-flight transfer.
-- The cleanup service + application code must delete rows immediately after
-- transfer completes (success or failure). There are no soft deletes here.
-- Audit trail: NONE by design. Transfer happened = no record remains.

CREATE TYPE transfer_status AS ENUM ('initiated', 'pending_accept', 'in_progress', 'completed', 'failed', 'expired');

CREATE TABLE transfers (
    id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    -- Who is sending
    sender_id       UUID NOT NULL REFERENCES spaces(id) ON DELETE CASCADE,
    -- Who is receiving (may be remote - in which case recipient_actor_url is used)
    recipient_id    UUID REFERENCES spaces(id) ON DELETE CASCADE,
    recipient_actor_url TEXT,   -- used when recipient is remote
    -- Encrypted payload (AES-256-GCM, key exchanged via ECDH)
    payload_enc     BYTEA,
    -- ECDH ephemeral public key used for this transfer
    ecdh_pubkey     BYTEA,
    -- Transfer acceptance token (one-time, given to recipient to claim)
    accept_token_hash TEXT NOT NULL UNIQUE,
    status          transfer_status NOT NULL DEFAULT 'initiated',
    -- Hard expiry: if not completed within this time, auto-deleted
    expires_at      TIMESTAMPTZ NOT NULL DEFAULT (NOW() + INTERVAL '15 minutes'),
    -- NO created_at kept after deletion - but needed during life of record
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
    -- NOTE: No updated_at by design - reduces metadata leakage
);

CREATE INDEX idx_transfers_sender ON transfers (sender_id);
CREATE INDEX idx_transfers_accept_token ON transfers (accept_token_hash);
CREATE INDEX idx_transfers_expires_at ON transfers (expires_at);

-- Cleanup function called by the Go cleanup service
CREATE OR REPLACE FUNCTION purge_expired_transfers() RETURNS INTEGER AS $$
DECLARE deleted_count INTEGER;
BEGIN
    DELETE FROM transfers WHERE expires_at < NOW() OR status IN ('completed', 'failed');
    GET DIAGNOSTICS deleted_count = ROW_COUNT;
    RETURN deleted_count;
END;
$$ LANGUAGE plpgsql;
```

```sql
    retry_count     INTEGER NOT NULL DEFAULT 0,
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    -- Auto-purge after processing
    process_after   TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_ap_inbox_target ON ap_inbox_queue (target_space_id);
CREATE INDEX idx_ap_inbox_processed ON ap_inbox_queue (processed, process_after);

CREATE TABLE ap_outbox_queue (
    id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    -- The local SPACE sending this
    sender_space_id UUID NOT NULL REFERENCES spaces(id) ON DELETE CASCADE,
    -- Target inbox URL
    target_inbox_url TEXT NOT NULL,
    -- The activity JSON to POST
    raw_activity    JSONB NOT NULL,
    status          ap_delivery_status NOT NULL DEFAULT 'pending',
    retry_count     INTEGER NOT NULL DEFAULT 0,
    next_retry_at   TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_ap_outbox_sender ON ap_outbox_queue (sender_space_id);
CREATE INDEX idx_ap_outbox_status ON ap_outbox_queue (status, next_retry_at);

-- Purge successfully delivered outbox items after 1 hour
-- (enough time to confirm delivery, then gone)
CREATE OR REPLACE FUNCTION purge_delivered_ap_activities() RETURNS INTEGER AS $$
DECLARE deleted_count INTEGER;
BEGIN
    DELETE FROM ap_outbox_queue
    WHERE status = 'delivered' AND created_at < NOW() - INTERVAL '1 hour';
    DELETE FROM ap_inbox_queue
    WHERE processed = true AND created_at < NOW() - INTERVAL '1 hour';
    GET DIAGNOSTICS deleted_count = ROW_COUNT;
    RETURN deleted_count;
END;
$$ LANGUAGE plpgsql;
```

```go
// SignRequest adds HTTP Signature headers to an outbound ActivityPub request.
// Implements the Mastodon-compatible signing scheme used by all major AP servers.
func SignRequest(req *http.Request, keyID string, privKey *rsa.PrivateKey, body []byte) error {
    date := time.Now().UTC().Format(http.TimeFormat)
    req.Header.Set("Date", date)
    
    // Digest header: SHA-256 of body
    h := sha256.Sum256(body)
    digest := "SHA-256=" + base64.StdEncoding.EncodeToString(h[:])
    req.Header.Set("Digest", digest)
    
    // Build signing string
    headers := "(request-target) host date digest"
    signingString := fmt.Sprintf(
        "(request-target): post %s\nhost: %s\ndate: %s\ndigest: %s",
        req.URL.Path, req.URL.Host, date, digest,
    )
    
    // Sign with RSA-SHA256
    hashed := sha256.Sum256([]byte(signingString))
    sig, err := rsa.SignPKCS1v15(rand.Reader, privKey, crypto.SHA256, hashed[:])
    if err != nil { return err }
    
    sigB64 := base64.StdEncoding.EncodeToString(sig)
    req.Header.Set("Signature", fmt.Sprintf(
        `keyId="%s",algorithm="rsa-sha256",headers="%s",signature="%s"`,
        keyID, headers, sigB64,
    ))
    return nil
}
```

```go
package middleware

import (

    "github.com/golang-jwt/jwt/v5"

type contextKey string
const SpaceIDKey contextKey = "space_id"

type Claims struct {
    SpaceID string `json:"space_id"`
    Handle  string `json:"handle"`
    jwt.RegisteredClaims
}

func RequireAuth(jwtSecret string) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            auth := r.Header.Get("Authorization")
            if !strings.HasPrefix(auth, "Bearer ") {
                http.Error(w, `{"error":"unauthorized"}`, http.StatusUnauthorized)
                return
            }
            tokenStr := strings.TrimPrefix(auth, "Bearer ")
            
            claims := &Claims{}
            token, err := jwt.ParseWithClaims(tokenStr, claims, func(t *jwt.Token) (interface{}, error) {
                if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
                    return nil, fmt.Errorf("unexpected signing method")
                }
                return []byte(jwtSecret), nil
            })
            
            if err != nil || !token.Valid {
                http.Error(w, `{"error":"invalid_token"}`, http.StatusUnauthorized)
                return
            }
            
            ctx := context.WithValue(r.Context(), SpaceIDKey, claims.SpaceID)
            next.ServeHTTP(w, r.WithContext(ctx))
        })
    }
}
```

```go
// POST /api/v1/transfers
// Initiates an ephemeral data transfer. Creates a one-time token.
func (h *TransferHandler) Initiate(w http.ResponseWriter, r *http.Request) {
    spaceID := r.Context().Value(middleware.SpaceIDKey).(string)
    
    var req struct {
        RecipientHandle    string `json:"recipient_handle"`
        RecipientActorURL  string `json:"recipient_actor_url"` // for remote spaces
        Payload            []byte `json:"payload"`             // already encrypted client-side
        PayloadType        string `json:"payload_type"`        // e.g. "profile_export"
    }
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, `{"error":"invalid_request"}`, http.StatusBadRequest)
        return
    }
    
    token, expiresAt, err := h.svc.InitiateTransfer(r.Context(), spaceID, req)
    if err != nil {
        http.Error(w, `{"error":"transfer_failed"}`, http.StatusInternalServerError)
        return
    }
    
    // Return one-time token. Client sends this to recipient out-of-band.
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(map[string]interface{}{
        "token":      token,
        "expires_at": expiresAt,
    })
}

// POST /api/v1/transfers/{token}/complete
// Called by recipient after consuming payload. Immediately deletes transfer record.
func (h *TransferHandler) Complete(w http.ResponseWriter, r *http.Request) {
    spaceID := r.Context().Value(middleware.SpaceIDKey).(string)
    token := chi.URLParam(r, "token")
    
    if err := h.svc.CompleteAndDelete(r.Context(), spaceID, token); err != nil {
        http.Error(w, `{"error":"complete_failed"}`, http.StatusInternalServerError)
        return
    }
    
    // 204: transfer is gone, no trace
    w.WriteHeader(http.StatusNoContent)
}
```

```go
    }

}
```

```go
    if err := db.RunMigrations(cfg.DatabaseURL); err != nil {
        fmt.Fprintf(os.Stderr, "migrations: %v\n", err); os.Exit(1)
    }
    
    // Init connections
    pool, err := db.NewPool(context.Background(), cfg.DatabaseURL)
    if err != nil { fmt.Fprintf(os.Stderr, "db: %v\n", err); os.Exit(1) }
    defer pool.Close()
    
    redisClient, err := cache.NewClient(cfg.RedisURL)
    if err != nil { fmt.Fprintf(os.Stderr, "redis: %v\n", err); os.Exit(1) }
    defer redisClient.Close()
    
    cryptoSvc := crypto.NewService([]byte(cfg.EncryptionKey))
    queries := dbsqlc.New(pool)
    
    // Build services
    authSvc    := services.NewAuthService(queries, redisClient, cfg, cryptoSvc)
    spaceSvc   := services.NewSpaceService(queries, cryptoSvc, cfg)
    transferSvc := services.NewTransferService(queries, cryptoSvc)
    connSvc    := services.NewConnectionService(queries)
    fedSvc     := services.NewFederationService(queries, redisClient, cryptoSvc, cfg)
    cleanupSvc := services.NewCleanupService(queries)
    
    // Build handlers
    deps := &router.Dependencies{
        Config: cfg,
        Handlers: &router.Handlers{
            Auth:        handlers.NewAuthHandler(authSvc),
            Spaces:      handlers.NewSpacesHandler(spaceSvc),
            Transfers:   handlers.NewTransferHandler(transferSvc),
            Connections: handlers.NewConnectionHandler(connSvc),
            ActivityPub: handlers.NewAPHandler(fedSvc, queries, cfg),
            WellKnown:   handlers.NewWellKnownHandler(queries, cfg),
            Health:      handlers.NewHealthHandler(pool, redisClient),
        },
        Middleware: &router.Middleware{
            Auth:    middleware.RequireAuth(cfg.JWTSecret),
            HTTPSig: middleware.VerifyHTTPSignature(queries),
        },
    }
    
    r := router.New(deps)
    
    // Start background cleanup (privacy enforcement)
    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()
    cleanupSvc.Start(ctx)
    
    // Also start the AP inbox processor
    go fedSvc.StartInboxProcessor(ctx)
    
    // Start federated delivery worker
    go fedSvc.StartDeliveryWorker(ctx)
    
    srv := &http.Server{
        Addr:         fmt.Sprintf("%s:%d", cfg.Host, cfg.Port),
        Handler:      r,
        ReadTimeout:  15 * time.Second,
        WriteTimeout: 30 * time.Second,
        IdleTimeout:  60 * time.Second,
    }
    
    // Graceful shutdown
    go func() {
        sig := make(chan os.Signal, 1)
        signal.Notify(sig, syscall.SIGINT, syscall.SIGTERM)
        <-sig
        shutCtx, shutCancel := context.WithTimeout(context.Background(), 30*time.Second)
        defer shutCancel()
        srv.Shutdown(shutCtx)
    }()
    
    fmt.Printf("ENGINE running on %s\n", srv.Addr)
    if err := srv.ListenAndServe(); err != http.ErrServerClosed {
        fmt.Fprintf(os.Stderr, "server: %v\n", err)
    }
}
```

## Key Architectural Decisions Justified

**Why sqlc over GORM for transfers:** The `DELETE FROM transfers WHERE id = $1` in `transfers.sql.go` is exactly what sqlc generates - you can read the exact SQL. With GORM you'd need to explicitly call `.Unscoped().Delete()` and remember to never enable soft-delete on this model. The privacy guarantee is auditable at the SQL layer.

**Why Chi over Gin/Echo/Fiber:** The `/.well-known/webfinger` and `/users/{handle}/inbox` routes need to be on the same domain for ActivityPub to work. Chi's sub-router mounting handles this cleanly. Gin and Echo use slightly different middleware composition patterns that require wrapping when adding HTTP Signature verification to specific routes. Fiber uses fasthttp which has subtle incompatibilities with the standard `http.Handler` interface that several ActivityPub libraries assume.

**Why scratch Docker image:** Engine will be a federated server that remote servers will POST to. Minimizing the attack surface of the runtime image is a direct security benefit. The binary is completely static (CGO disabled), so scratch works perfectly.

**Why Ed25519 JWT but RSA-4096 for ActivityPub:** JWT tokens are issued and verified by the same server (symmetric HMAC-SHA256 is simplest and fastest for this). ActivityPub HTTP Signatures must be RSA because Mastodon, Lemmy, and every other AP implementation expects RSA-SHA256 for cross-server verification. This is a compatibility constraint, not a preference.

**Why embed migrations in binary:** Running `go run main.go` or the Docker container both automatically migrate the database to the latest schema. There's no separate migration step in CI/CD or deployment. The `golang-migrate` library with `iofs.New(migrationsFS, "migrations")` makes this trivially safe.

# ENGINE Platform - Complete Implementation Plan

## Architectural Decision Log

### Framework Choice: Chi (not Gin, Echo, or Fiber)

Chi is the right choice for ENGINE because:
- It is 100% `net/http`-compatible, meaning every middleware written for `net/http` works unchanged. ActivityPub requires precise HTTP signature verification, and standard `net/http` handlers make that trivial.
- It has zero mandatory dependencies in the router itself — the router module is all you import for routing. Optional middleware lives in separate packages.
- It is the fastest to learn and the most maintainable long-term for a system that will expand to 8+ services. Gin's middleware interface is non-standard; Fiber's is incompatible with `net/http` (uses fasthttp). Echo is fine but Chi's composability is better.
- WebSocket support via `nhooyr.io/websocket` or `gorilla/websocket` works directly since handlers are plain `http.Handler`.

### ORM Choice: sqlc (not GORM)

For a privacy-first system where query behavior must be auditable and exact:
- GORM silently issues extra queries, can leak data via preloads, and makes it hard to reason about what SQL runs.
- sqlc generates type-safe Go code from raw SQL. You write the migration SQL, write query SQL, run `sqlc generate`, and get compile-time-checked Go functions. Every query is visible in source control. This perfectly matches the "no logs, no leakage" model.
- For ephemeral transfers (which must be permanently deleted), you want exact `DELETE` statements, not ORM magic.

### Monorepo: Go Workspace + pnpm Workspaces

- `go.work` file at root coordinates multiple Go modules (one per backend service).
- `pnpm-workspace.yaml` at root coordinates all frontend packages.
- Each platform (ENGINE, MYSPACE, BOARD, etc.) is its own Go module under `services/` and its own React app under `apps/`.

### Phase 4: Database Schema (Migration Files)

**Critical design decisions for privacy:**
- No IP addresses stored anywhere.
- `transfers` table auto-deletes on completion using PostgreSQL `RETURNING` + immediate DELETE in a transaction.
- Sessions use opaque token hashes (SHA-256 of the actual token), never the token itself.
- All `created_at` timestamps are present for data integrity but no operational logs are persisted.

`/home/user/nTogether/services/engine/db/migrations/001_create_spaces.sql`:
```sql
-- A SPACE is the root identity unit. One per user across all nTogether platforms.
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE TABLE spaces (
    id            UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    handle        VARCHAR(64) UNIQUE NOT NULL,           -- @username (globally unique)
    display_name  VARCHAR(128) NOT NULL,
    email_hash    BYTEA NOT NULL UNIQUE,                 -- SHA-256 of email, never plaintext
    password_hash TEXT NOT NULL,                         -- bcrypt
    public_key    TEXT NOT NULL,                         -- RSA public key (PEM) for AP + signing
    private_key   TEXT NOT NULL,                         -- RSA private key (PEM, stored encrypted)
    is_active     BOOLEAN NOT NULL DEFAULT true,
    is_local      BOOLEAN NOT NULL DEFAULT true,         -- false = federated remote space
    created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_spaces_handle ON spaces(handle);
CREATE INDEX idx_spaces_is_local ON spaces(is_local);

-- Trigger to auto-update updated_at
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER spaces_updated_at
    BEFORE UPDATE ON spaces
    FOR EACH ROW EXECUTE FUNCTION update_updated_at();
```

```sql
-- Connections represent follow/link relationships between SPACEs.
-- Supports both local (same ENGINE) and federated (AP) connections.
CREATE TYPE connection_status AS ENUM ('pending', 'accepted', 'rejected', 'blocked');
CREATE TYPE connection_type AS ENUM ('follow', 'friend', 'block');

CREATE TABLE connections (
    id             UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    from_space_id  UUID NOT NULL REFERENCES spaces(id) ON DELETE CASCADE,
    to_space_id    UUID,                                 -- NULL if remote federated
    to_ap_actor    TEXT,                                 -- AP actor URI if federated
    conn_type      connection_type NOT NULL DEFAULT 'follow',
    status         connection_status NOT NULL DEFAULT 'pending',
    created_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at     TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    -- One connection per pair per type
    CONSTRAINT unique_connection UNIQUE(from_space_id, to_space_id, conn_type),
    CONSTRAINT check_target CHECK (
        (to_space_id IS NOT NULL AND to_ap_actor IS NULL) OR
        (to_space_id IS NULL AND to_ap_actor IS NOT NULL)
    )
);

CREATE INDEX idx_connections_from ON connections(from_space_id);
CREATE INDEX idx_connections_to ON connections(to_space_id);
CREATE INDEX idx_connections_status ON connections(status);
```

`/home/user/nTogether/services/engine/db/migrations/004_create_transfers.sql`:
```sql
-- Ephemeral transfer records. These exist ONLY while a transfer is in-flight.
-- On completion (success or failure), the row is immediately deleted.
-- The transfer data payload is encrypted with the recipient's public key.
CREATE TYPE transfer_status AS ENUM ('initiated', 'in_progress', 'completed', 'failed');

CREATE TABLE transfers (
    id                UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    from_space_id     UUID NOT NULL REFERENCES spaces(id) ON DELETE CASCADE,
    to_space_id       UUID NOT NULL REFERENCES spaces(id) ON DELETE CASCADE,
    encrypted_payload TEXT NOT NULL,                     -- AES-256-GCM encrypted, base64
    payload_hash      BYTEA NOT NULL,                    -- SHA-256 of plaintext for integrity
    status            transfer_status NOT NULL DEFAULT 'initiated',
    expires_at        TIMESTAMPTZ NOT NULL DEFAULT (NOW() + INTERVAL '10 minutes'),
    created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW()
    -- NOTE: NO updated_at - we never want partial history of a transfer
);

-- Transfers expire after 10 minutes if not completed
CREATE INDEX idx_transfers_expires ON transfers(expires_at);
CREATE INDEX idx_transfers_from ON transfers(from_space_id);
CREATE INDEX idx_transfers_to ON transfers(to_space_id);
CREATE INDEX idx_transfers_status ON transfers(status);

-- Cleanup job: auto-delete expired transfers
-- This is also enforced in application code, but belt-and-suspenders
CREATE OR REPLACE FUNCTION delete_expired_transfers()
RETURNS void AS $$
BEGIN
    DELETE FROM transfers WHERE expires_at < NOW();
END;
$$ LANGUAGE plpgsql;
```

```go
func (s *Server) Start(ctx context.Context) error {
    s.logger.Info("engine starting", "addr", s.cfg.Addr(), "domain", s.cfg.Domain)

    errCh := make(chan error, 1)
    go func() {
        if err := s.http.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
            errCh <- err

        s.logger.Info("shutting down engine")
        shutCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)

        return s.http.Shutdown(shutCtx)
    }
```

```go
package middleware

import (
    "context"
    "net/http"
    "strings"

    "ntogether/engine/internal/httputil"
    "ntogether/engine/internal/services"
)

type contextKey string

const SpaceIDKey contextKey = "space_id"

func RequireAuth(authSvc *services.AuthService) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            header := r.Header.Get("Authorization")
            if header == "" || !strings.HasPrefix(header, "Bearer ") {
                httputil.Unauthorized(w, "missing authorization header")
                return
            }

            token := strings.TrimPrefix(header, "Bearer ")
            spaceID, err := authSvc.ValidateAccessToken(token)
            if err != nil {
                httputil.Unauthorized(w, "invalid or expired token")
                return
            }

            ctx := context.WithValue(r.Context(), SpaceIDKey, spaceID)
            next.ServeHTTP(w, r.WithContext(ctx))
        })
    }
}

func SpaceIDFromContext(ctx context.Context) (string, bool) {
    id, ok := ctx.Value(SpaceIDKey).(string)
    return id, ok
}
```

```go
// EphemeralLogger logs request metadata transiently.
// The log entry exists only in memory during the request lifecycle;
// no persistent log storage is used.
func EphemeralLogger(logger *slog.Logger) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            start := time.Now()
            ww := middleware.NewWrapResponseWriter(w, r.ProtoMajor)

            defer func() {
                // Log to stdout only (no file, no persistent store)
                // In production, configure log aggregator to NOT persist these.
                logger.Info("request",
                    "method", r.Method,
                    "path", r.URL.Path,
                    "status", ww.Status(),
                    "duration_ms", time.Since(start).Milliseconds(),
                    "request_id", middleware.GetReqID(r.Context()),
                    // Deliberately omit: IP, User-Agent, query params
                )
            }()

            next.ServeHTTP(ww, r)
        })
    }
}
```

```go
package services

import (

    "encoding/base64"
    "encoding/pem"
    "fmt"
    "os"

    "golang.org/x/crypto/bcrypt"

    "ntogether/engine/config"
)

type CryptoService struct {
    privateKey *rsa.PrivateKey
    publicKey  *rsa.PublicKey
    publicKeyPEM string
}

func NewCryptoService(cfg *config.Config) *CryptoService {
    priv, pub, pubPEM := loadOrGenerateKeys(cfg.APPrivateKeyPath, cfg.APPublicKeyPath)
    return &CryptoService{
        privateKey:   priv,
        publicKey:    pub,
        publicKeyPEM: pubPEM,
    }
}

// GenerateSpaceKeyPair generates an RSA-2048 key pair for a new SPACE.
// The private key is encrypted with the space's password-derived key before storage.
func (s *CryptoService) GenerateSpaceKeyPair() (publicKeyPEM, privateKeyPEM string, err error) {
    key, err := rsa.GenerateKey(rand.Reader, 2048)
    if err != nil {
        return "", "", fmt.Errorf("generate key: %w", err)
    }

    privBytes, err := x509.MarshalPKCS8PrivateKey(key)
    if err != nil {
        return "", "", fmt.Errorf("marshal private key: %w", err)
    }

    pubBytes, err := x509.MarshalPKIXPublicKey(&key.PublicKey)
    if err != nil {
        return "", "", fmt.Errorf("marshal public key: %w", err)
    }

    privPEM := pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: privBytes})
    pubPEM := pem.EncodeToMemory(&pem.Block{Type: "PUBLIC KEY", Bytes: pubBytes})

    return string(pubPEM), string(privPEM), nil
}

// HashEmail creates a one-way hash of an email for storage.
// We never store plaintext emails.
func (s *CryptoService) HashEmail(email string) []byte {
    h := sha256.Sum256([]byte(email))
    return h[:]
}

// HashPassword creates a bcrypt hash of a password.
func (s *CryptoService) HashPassword(password string) (string, error) {
    b, err := bcrypt.GenerateFromPassword([]byte(password), bcrypt.DefaultCost)
    return string(b), err
}

// VerifyPassword checks a password against its bcrypt hash.
func (s *CryptoService) VerifyPassword(hash, password string) error {
    return bcrypt.CompareHashAndPassword([]byte(hash), []byte(password))
}

// EncryptForRecipient encrypts data using a recipient's RSA public key (for transfers).
func (s *CryptoService) EncryptForRecipient(data []byte, recipientPublicPEM string) (string, error) {
    block, _ := pem.Decode([]byte(recipientPublicPEM))
    if block == nil {
        return "", fmt.Errorf("invalid public key PEM")
    }

    pub, err := x509.ParsePKIXPublicKey(block.Bytes)
    if err != nil {
        return "", fmt.Errorf("parse public key: %w", err)
    }

    rsaPub, ok := pub.(*rsa.PublicKey)
    if !ok {
        return "", fmt.Errorf("not an RSA public key")
    }

    encrypted, err := rsa.EncryptOAEP(sha256.New(), rand.Reader, rsaPub, data, nil)
    if err != nil {
        return "", fmt.Errorf("encrypt: %w", err)
    }

    return base64.StdEncoding.EncodeToString(encrypted), nil
}

// Sign signs data with the ENGINE's private key (for ActivityPub HTTP signatures).
func (s *CryptoService) Sign(data []byte) ([]byte, error) {
    h := sha256.Sum256(data)
    return rsa.SignPKCS1v15(rand.Reader, s.privateKey, crypto.SHA256, h[:])
}

func (s *CryptoService) PublicKeyPEM() string {
    return s.publicKeyPEM
}

func loadOrGenerateKeys(privPath, pubPath string) (*rsa.PrivateKey, *rsa.PublicKey, string) {
    privPEM, err := os.ReadFile(privPath)
    if err != nil {
        // Generate new keys if not found
        key, _ := rsa.GenerateKey(rand.Reader, 2048)
        return key, &key.PublicKey, ""
    }
```

```go
    pwHash, err := s.crypto.HashPassword(password)

    space, err := s.spaceStore.GetByEmailHash(ctx, emailHash)
    if err != nil {
        return nil, fmt.Errorf("invalid credentials")
    }

    if err := s.crypto.VerifyPassword(space.PasswordHash, password); err != nil {
        return nil, fmt.Errorf("invalid credentials")
    }

    return s.issueTokenPair(ctx, space.ID.String(), space.Handle)
}

func (s *AuthService) Refresh(ctx context.Context, refreshToken string) (*TokenPair, error) {
    hash := hashToken(refreshToken)
    session, err := s.sessionStore.GetByTokenHash(ctx, hash)
    if err != nil {
        return nil, fmt.Errorf("invalid refresh token")
    }

    // Rotate: delete old, issue new
    newRefreshToken := uuid.New().String() + uuid.New().String()
    newHash := hashToken(newRefreshToken)
    newExpiry := time.Now().Add(s.cfg.JWTRefreshExpiry)

    if _, err := s.sessionStore.Rotate(ctx, hash, newHash, newExpiry); err != nil {
        return nil, fmt.Errorf("rotate session: %w", err)
    }

    space, err := s.spaceStore.GetByID(ctx, session.SpaceID.String())
    if err != nil {
        return nil, fmt.Errorf("space not found")
    }

    access, err := s.generateAccessToken(space.ID.String(), space.Handle)
    if err != nil {
        return nil, err
    }

    return &TokenPair{
        AccessToken:  access,
        RefreshToken: newRefreshToken,
        ExpiresIn:    int(s.cfg.JWTAccessExpiry.Seconds()),
    }, nil
}

    hash := hashToken(refreshToken)
    return s.sessionStore.Delete(ctx, hash)
}

func (s *AuthService) ValidateAccessToken(token string) (string, error) {
    claims := &spaceClaims{}
    parsed, err := jwt.ParseWithClaims(token, claims, func(t *jwt.Token) (interface{}, error) {
        if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
            return nil, fmt.Errorf("unexpected signing method")
        }
        return []byte(s.cfg.JWTSecret), nil
    })
    if err != nil || !parsed.Valid {
        return "", fmt.Errorf("invalid token")
    }
    return claims.SpaceID, nil
}

func (s *AuthService) issueTokenPair(ctx context.Context, spaceID, handle string) (*TokenPair, error) {
    access, err := s.generateAccessToken(spaceID, handle)
    if err != nil {
        return nil, err
    }

    refreshToken := uuid.New().String() + uuid.New().String()
    hash := hashToken(refreshToken)
    expiry := time.Now().Add(s.cfg.JWTRefreshExpiry)

    if _, err := s.sessionStore.Create(ctx, spaceID, hash, expiry); err != nil {
        return nil, fmt.Errorf("create session: %w", err)
    }

    return &TokenPair{
        AccessToken:  access,
        RefreshToken: refreshToken,
        ExpiresIn:    int(s.cfg.JWTAccessExpiry.Seconds()),
    }, nil
}

func (s *AuthService) generateAccessToken(spaceID, handle string) (string, error) {
    now := time.Now()
    claims := spaceClaims{
        SpaceID: spaceID,
        Handle:  handle,
        RegisteredClaims: jwt.RegisteredClaims{
            IssuedAt:  jwt.NewNumericDate(now),
            ExpiresAt: jwt.NewNumericDate(now.Add(s.cfg.JWTAccessExpiry)),
            Issuer:    "ntogether-engine",
        },
    }
    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    return token.SignedString([]byte(s.cfg.JWTSecret))
}

func hashToken(token string) []byte {
    h := sha256.Sum256([]byte(token))
    _ = hex.EncodeToString(h[:]) // for readability in debugging only
    return h[:]
```

```go
type TransferService struct {
    transferStore store.TransferStore
    spaceStore    store.SpaceStore
    crypto        *CryptoService
    cache         *CacheService
}

type TransferResult struct {
    TransferID string    `json:"transfer_id"`
    ExpiresAt  time.Time `json:"expires_at"`
    Status     string    `json:"status"`
}

func NewTransferService(ts store.TransferStore, ss store.SpaceStore, cs *CryptoService, ca *CacheService) *TransferService {
    return &TransferService{transferStore: ts, spaceStore: ss, crypto: cs, cache: ca}
}

// Initiate creates an encrypted transfer record. The payload is encrypted with
// the recipient's public key so only they can decrypt it.
func (s *TransferService) Initiate(ctx context.Context, fromSpaceID, toHandle string, payload []byte) (*TransferResult, error) {
    toSpace, err := s.spaceStore.GetByHandle(ctx, toHandle)
    if err != nil {
        return nil, fmt.Errorf("recipient not found")
    }

    // Encrypt payload with recipient's public key
    encrypted, err := s.crypto.EncryptForRecipient(payload, toSpace.PublicKey)
    if err != nil {
        return nil, fmt.Errorf("encrypt payload: %w", err)
    }

    // Hash the original payload for integrity verification
    hash := sha256.Sum256(payload)

    transfer, err := s.transferStore.Create(ctx, store.CreateTransferParams{
        FromSpaceID:      fromSpaceID,
        ToSpaceID:        toSpace.ID.String(),
        EncryptedPayload: encrypted,
        PayloadHash:      hash[:],
    })
    if err != nil {
        return nil, fmt.Errorf("create transfer: %w", err)
    }

    // Notify recipient via Redis pub/sub (WebSocket fan-out)
    s.cache.Publish(ctx, "space:"+toSpace.ID.String()+":events", map[string]string{
        "type":        "transfer_incoming",
        "transfer_id": transfer.ID.String(),
    })

    return &TransferResult{
        TransferID: transfer.ID.String(),
        ExpiresAt:  transfer.ExpiresAt,
        Status:     "initiated",
    }, nil
}

// Accept retrieves, decrypts, and PERMANENTLY DELETES the transfer record.
// After this call, the transfer no longer exists anywhere in the system.
func (s *TransferService) Accept(ctx context.Context, transferID, toSpaceID string, privateKeyPEM string) ([]byte, error) {
    // This is a single atomic DELETE ... RETURNING query
    transfer, err := s.transferStore.CompleteAndDelete(ctx, transferID, toSpaceID)
    if err != nil {
        return nil, fmt.Errorf("transfer not found or already completed")
    }

    // Decrypt with recipient's private key (provided in-request, never stored)
    decrypted, err := decryptWithPrivateKey(transfer.EncryptedPayload, privateKeyPEM)
    if err != nil {
        return nil, fmt.Errorf("decrypt failed: %w", err)
    }

    // Verify integrity
    hash := sha256.Sum256(decrypted)
    if !bytesEqual(hash[:], transfer.PayloadHash) {
        return nil, fmt.Errorf("payload integrity check failed")
    }

    return decrypted, nil
}

// Cancel immediately deletes a transfer the sender initiated.
func (s *TransferService) Cancel(ctx context.Context, transferID, fromSpaceID string) error {
    return s.transferStore.DeleteFailed(ctx, transferID)
}
```

```go
import (
    "crypto"
    "crypto/rand"
    "crypto/rsa"
    "crypto/sha256"
    "crypto/x509"
    "encoding/base64"
    "encoding/pem"
    "fmt"
    "net/http"
    "strings"
    "time"
)

// PublicKeyFetcher retrieves a remote actor's public key by key ID URL.
type PublicKeyFetcher func(keyID string) (*rsa.PublicKey, error)

// SignRequest adds an HTTP Signature to an outgoing ActivityPub request.
// This is required for federation - receiving servers verify the signature.
func SignRequest(req *http.Request, keyID string, privateKey *rsa.PrivateKey) error {
    date := time.Now().UTC().Format(http.TimeFormat)
    req.Header.Set("Date", date)
    req.Header.Set("Host", req.URL.Host)

    signedHeaders := "(request-target) host date"
    signingString := fmt.Sprintf(
        "(request-target): %s %s\nhost: %s\ndate: %s",
        strings.ToLower(req.Method),
        req.URL.RequestURI(),
        req.URL.Host,
        date,
    )

    h := sha256.Sum256([]byte(signingString))
    sig, err := rsa.SignPKCS1v15(rand.Reader, privateKey, crypto.SHA256, h[:])
    if err != nil {
        return fmt.Errorf("sign: %w", err)
    }

    sigB64 := base64.StdEncoding.EncodeToString(sig)
    req.Header.Set("Signature", fmt.Sprintf(
        `keyId="%s",algorithm="rsa-sha256",headers="%s",signature="%s"`,
        keyID, signedHeaders, sigB64,
    ))

    return nil
}

// VerifyRequest verifies the HTTP Signature on an incoming ActivityPub request.
func VerifyRequest(req *http.Request, fetchKey PublicKeyFetcher) error {
    sigHeader := req.Header.Get("Signature")
    if sigHeader == "" {
        return fmt.Errorf("missing Signature header")
    }

    params := parseSignatureHeader(sigHeader)
    keyID := params["keyId"]
    sigB64 := params["signature"]
    headers := params["headers"]

    if keyID == "" || sigB64 == "" {
        return fmt.Errorf("malformed Signature header")
    }

    pubKey, err := fetchKey(keyID)
    if err != nil {
        return fmt.Errorf("fetch public key %s: %w", keyID, err)
    }

    signingString := buildSigningString(req, strings.Fields(headers))
    h := sha256.Sum256([]byte(signingString))

    sigBytes, err := base64.StdEncoding.DecodeString(sigB64)
    if err != nil {
        return fmt.Errorf("decode signature: %w", err)
    }

    return rsa.VerifyPKCS1v15(pubKey, crypto.SHA256, h[:], sigBytes)
}

func parseSignatureHeader(header string) map[string]string {
    result := make(map[string]string)
    for _, part := range strings.Split(header, ",") {
        kv := strings.SplitN(strings.TrimSpace(part), "=", 2)
        if len(kv) == 2 {
            key := strings.TrimSpace(kv[0])
            val := strings.Trim(strings.TrimSpace(kv[1]), `"`)
            result[key] = val
        }
    }
    return result
}

func buildSigningString(req *http.Request, headers []string) string {
    parts := make([]string, 0, len(headers))
    for _, h := range headers {
        switch h {
        case "(request-target)":
            parts = append(parts, fmt.Sprintf("(request-target): %s %s",
                strings.ToLower(req.Method), req.URL.RequestURI()))
        default:
            parts = append(parts, fmt.Sprintf("%s: %s", h, req.Header.Get(h)))
        }
    }
    return strings.Join(parts, "\n")
}

func ParseRSAPublicKey(pemStr string) (*rsa.PublicKey, error) {
    block, _ := pem.Decode([]byte(pemStr))
    if block == nil {
        return nil, fmt.Errorf("invalid PEM block")
    }
    pub, err := x509.ParsePKIXPublicKey(block.Bytes)
    if err != nil {
        return nil, err
    }
    rsaPub, ok := pub.(*rsa.PublicKey)
    if !ok {
        return nil, fmt.Errorf("not RSA public key")
    }
    return rsaPub, nil
}
```

```go
package handlers

import (
    "log/slog"
    "net/http"
    "sync"

    "github.com/gorilla/websocket"

    "ntogether/engine/internal/middleware"
)

var upgrader = websocket.Upgrader{
    ReadBufferSize:  1024,
    WriteBufferSize: 1024,
    CheckOrigin: func(r *http.Request) bool {
        return true // CORS handled by middleware
    },
}

type WSClient struct {
    spaceID string
    conn    *websocket.Conn
    send    chan []byte
}

type WSHub struct {
    mu      sync.RWMutex
    clients map[string]*WSClient // spaceID -> client
    logger  *slog.Logger
}

type WebSocketHandler struct {
    hub *WSHub
}

func NewWebSocketHandler(logger *slog.Logger) *WebSocketHandler {
    hub := &WSHub{
        clients: make(map[string]*WSClient),
        logger:  logger,
    }
    return &WebSocketHandler{hub: hub}
}

func (h *WebSocketHandler) Handle(w http.ResponseWriter, r *http.Request) {
    spaceID, ok := middleware.SpaceIDFromContext(r.Context())
    if !ok {
        http.Error(w, "unauthorized", http.StatusUnauthorized)
        return
    }

    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil {
        h.hub.logger.Error("ws upgrade failed", "error", err)
        return
    }

    client := &WSClient{
        spaceID: spaceID,
        conn:    conn,
        send:    make(chan []byte, 256),
    }

    h.hub.mu.Lock()
    h.hub.clients[spaceID] = client
    h.hub.mu.Unlock()

    go client.writePump(h.hub)
    go client.readPump(h.hub)
}

func (c *WSClient) writePump(hub *WSHub) {
    defer func() {
        c.conn.Close()
        hub.mu.Lock()
        delete(hub.clients, c.spaceID)
        hub.mu.Unlock()
    }()

    for msg := range c.send {
        if err := c.conn.WriteMessage(websocket.TextMessage, msg); err != nil {
            return
        }
    }
}

func (c *WSClient) readPump(hub *WSHub) {
    defer close(c.send)
    for {
        _, _, err := c.conn.ReadMessage()
        if err != nil {
            return
        }
        // Client-to-server messages not needed for now
    }
}

// Broadcast sends a message to a specific SPACE's connected WebSocket client.
func (h *WSHub) Broadcast(spaceID string, msg []byte) {
    h.mu.RLock()
    client, ok := h.clients[spaceID]
    h.mu.RUnlock()

    if ok {
        select {
        case client.send <- msg:
        default:
            // Client's send buffer full; drop the message
        }
    }
}
```

```ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import axios from 'axios'

interface AuthState {
  accessToken: string | null
  refreshToken: string | null
  spaceID: string | null
  handle: string | null
  isAuthenticated: boolean
  login: (email: string, password: string) => Promise<void>
  register: (handle: string, email: string, password: string, displayName: string) => Promise<void>
  refresh: () => Promise<void>
  logout: () => void
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set, get) => ({
      accessToken: null,
      refreshToken: null,
      spaceID: null,
      handle: null,
      isAuthenticated: false,

      login: async (email, password) => {
        const res = await axios.post('/auth/login', { email, password })
        const { data } = res.data
        // Decode JWT to get spaceID and handle
        const payload = JSON.parse(atob(data.access_token.split('.')[1]))
        set({
          accessToken: data.access_token,
          refreshToken: data.refresh_token,
          spaceID: payload.space_id,
          handle: payload.handle,
          isAuthenticated: true,
        })
      },

      register: async (handle, email, password, displayName) => {
        const res = await axios.post('/auth/register', {
          handle, email, password, display_name: displayName,
        })
        const { data } = res.data
        const payload = JSON.parse(atob(data.access_token.split('.')[1]))
        set({
          accessToken: data.access_token,
          refreshToken: data.refresh_token,
          spaceID: payload.space_id,
          handle: payload.handle,
          isAuthenticated: true,
        })
      },

      refresh: async () => {
        const refreshToken = get().refreshToken
        if (!refreshToken) throw new Error('no refresh token')
        const res = await axios.post('/auth/refresh', { refresh_token: refreshToken })
        const { data } = res.data
        set({ accessToken: data.access_token, refreshToken: data.refresh_token })
      },

      logout: () => {
        const refreshToken = get().refreshToken
        if (refreshToken) {
          axios.post('/auth/logout', { refresh_token: refreshToken }).catch(() => {})
        }
        set({ accessToken: null, refreshToken: null, spaceID: null, handle: null, isAuthenticated: false })
      },
    }),
    {
      name: 'ntogether-auth',
      partialState: (state) => ({
        // Only persist refresh token - access token is short-lived and refreshed on load
        refreshToken: state.refreshToken,
        handle: state.handle,
        spaceID: state.spaceID,
      }),
    }
  )
)
```

```go
// ProcessInboxActivity handles an incoming ActivityPub activity.
func (s *FederationService) ProcessInboxActivity(ctx context.Context, targetHandle string, activity apub.Activity, r *http.Request) error {
    // Verify HTTP signature before processing
    if err := apub.VerifyRequest(r, s.fetchRemotePublicKey); err != nil {
        return fmt.Errorf("signature verification failed: %w", err)
    }

    switch activity.Type {
    case apub.TypeFollow:
        return s.handleFollow(ctx, targetHandle, activity)
    case apub.TypeUndo:
        return s.handleUndo(ctx, targetHandle, activity)
    case apub.TypeAccept:
        return s.handleAccept(ctx, targetHandle, activity)
    default:
        // Unknown activity types are silently ignored (AP spec requirement)
        return nil
    }
}

func (s *FederationService) handleFollow(ctx context.Context, targetHandle string, activity apub.Activity) error {
    actorURL := activity.Actor

    // Fetch remote actor to get their inbox
    remoteActor, err := s.FetchRemoteActor(ctx, actorURL)
    if err != nil {
        return fmt.Errorf("fetch remote actor: %w", err)
    }

    // Store the remote actor
    s.apStore.UpsertActor(ctx, store.UpsertActorParams{
        APID:              remoteActor.ID,
        ActorType:         remoteActor.Type,
        PreferredUsername: remoteActor.PreferredUsername,
        InboxURL:          remoteActor.Inbox,
        OutboxURL:         remoteActor.Outbox,
        PublicKey:         remoteActor.PublicKey.PublicKeyPem,
        RawData:           remoteActor,
    })

    // Create a pending connection (auto-accept for now; can add approval flow)
    s.connStore.Create(ctx, store.CreateConnectionParams{
        FromAPActor: actorURL,
        ToHandle:    targetHandle,
        ConnType:    "follow",
        Status:      "accepted",
    })

    // Send Accept activity back
    return s.SendAccept(ctx, targetHandle, activity)
}

// SendAccept delivers an Accept{Follow} activity to the remote actor's inbox.
func (s *FederationService) SendAccept(ctx context.Context, localHandle string, followActivity apub.Activity) error {
    base := fmt.Sprintf("https://%s", s.cfg.Domain)
    actorURL := fmt.Sprintf("%s/ap/users/%s", base, localHandle)

    accept := apub.Activity{
        Object: apub.Object{
            Context: apub.ActivityStreamsContext,
            ID:      fmt.Sprintf("%s/activities/accept/%d", actorURL, time.Now().UnixNano()),
            Type:    apub.TypeAccept,
        },
        Actor: actorURL,
        Obj:   followActivity,
    }

    remoteActor, err := s.FetchRemoteActor(ctx, followActivity.Actor)
    if err != nil {
        return err
    }

    return s.deliverActivity(ctx, accept, remoteActor.Inbox, localHandle)
}

// deliverActivity POSTs a signed ActivityPub activity to a remote inbox.
func (s *FederationService) deliverActivity(ctx context.Context, activity interface{}, inboxURL string, signingHandle string) error {
    body, err := json.Marshal(activity)
    if err != nil {
        return fmt.Errorf("marshal activity: %w", err)
    }

    req, err := http.NewRequestWithContext(ctx, http.MethodPost, inboxURL, bytes.NewReader(body))
    if err != nil {
        return err
    }
    req.Header.Set("Content-Type", "application/activity+json")

    keyID := fmt.Sprintf("https://%s/ap/users/%s#main-key", s.cfg.Domain, signingHandle)
    if err := apub.SignRequest(req, keyID, s.crypto.privateKey); err != nil {
        return fmt.Errorf("sign request: %w", err)
    }

    resp, err := s.client.Do(req)
    if err != nil {
        return fmt.Errorf("deliver: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode < 200 || resp.StatusCode > 299 {
        return fmt.Errorf("delivery failed with status %d", resp.StatusCode)
    }

    return nil
}

// FetchRemoteActor retrieves and caches an ActivityPub actor from a remote server.
func (s *FederationService) FetchRemoteActor(ctx context.Context, actorURL string) (*apub.Actor, error) {
    req, err := http.NewRequestWithContext(ctx, http.MethodGet, actorURL, nil)
    if err != nil {
        return nil, err
    }
    req.Header.Set("Accept", "application/activity+json")

    resp, err := s.client.Do(req)
    if err != nil {
        return nil, fmt.Errorf("fetch actor %s: %w", actorURL, err)
    }
    defer resp.Body.Close()

    var actor apub.Actor
    if err := json.NewDecoder(resp.Body).Decode(&actor); err != nil {
        return nil, fmt.Errorf("decode actor: %w", err)
    }

    return &actor, nil
}
```

```go
}

func (c *CacheService) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    b, err := json.Marshal(value)
    if err != nil {
        return err
    }
    return c.rdb.Set(ctx, key, b, ttl).Err()
}

func (c *CacheService) Get(ctx context.Context, key string, dest interface{}) error {
    b, err := c.rdb.Get(ctx, key).Bytes()
    if err != nil {
        return err
    }
    return json.Unmarshal(b, dest)
}

func (c *CacheService) Delete(ctx context.Context, key string) error {
    return c.rdb.Del(ctx, key).Err()
}

func (c *CacheService) Publish(ctx context.Context, channel string, msg interface{}) error {
    b, err := json.Marshal(msg)
    if err != nil {
        return fmt.Errorf("marshal pub message: %w", err)
    }
    return c.rdb.Publish(ctx, channel, b).Err()
}

func (c *CacheService) RateLimit(ctx context.Context, key string, max int, window time.Duration) (bool, error) {
    pipe := c.rdb.Pipeline()
    incr := pipe.Incr(ctx, key)
    pipe.Expire(ctx, key, window)
    _, err := pipe.Exec(ctx)
    if err != nil {
        return false, err
    }
    return incr.Val() <= int64(max), nil
}
```

```go
func (s *ConnectionService) Follow(ctx context.Context, fromSpaceID, toHandle string) (*store.Connection, error) {
    conn, err := s.connStore.Create(ctx, store.CreateConnectionParams{
        FromSpaceID: fromSpaceID,
        ToHandle:    toHandle,
        ConnType:    "follow",
        Status:      "pending",
    })
    if err != nil {
        return nil, fmt.Errorf("create follow: %w", err)
    }

    // Invalidate follower/following caches
    s.cache.Delete(ctx, "connections:following:"+fromSpaceID)

    return conn, nil
}

    if err != nil {

    }

    if err := s.connStore.Delete(ctx, conn.ID, fromSpaceID); err != nil {
        return fmt.Errorf("delete follow: %w", err)
    }

    s.cache.Delete(ctx, "connections:following:"+fromSpaceID)
    return nil
}

    return s.connStore.UpdateStatus(ctx, connectionID, "accepted")

    _, err := s.connStore.UpdateStatus(ctx, connectionID, "rejected")
    return err
```

```ts
import { useEffect, useRef, useCallback } from 'react'
import { useAuthStore } from '../store/auth'

type MessageHandler = (data: unknown) => void

export function useWebSocket(onMessage: MessageHandler) {
  const wsRef = useRef<WebSocket | null>(null)
  const token = useAuthStore((s) => s.accessToken)

  const connect = useCallback(() => {
    if (!token) return

    const proto = window.location.protocol === 'https:' ? 'wss' : 'ws'
    const ws = new WebSocket(
      `${proto}://${window.location.host}/ws`,
      // Pass token as subprotocol (standard WS auth pattern)
    )

    // Alternatively, append token as query param (simpler but less clean):
    // const ws = new WebSocket(`${proto}://${window.location.host}/ws?token=${token}`)

    ws.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data)
        onMessage(data)
      } catch {
        // ignore malformed messages
      }
    }

    ws.onclose = () => {
      // Reconnect after 3 seconds
      setTimeout(connect, 3000)
    }

    wsRef.current = ws
  }, [token, onMessage])
```

```go
// ProcessInboxActivity handles an incoming ActivityPub activity.

func (s *FederationService) handleFollow(ctx context.Context, targetHandle string, activity apub.Activity) error {

    s.apStore.UpsertActor(ctx, store.UpsertActorParams{

// deliverActivity POSTs a signed ActivityPub activity to a remote inbox.
func (s *FederationService) deliverActivity(ctx context.Context, activity interface{}, inboxURL string, signingHandle string) error {
    body, err := json.Marshal(activity)
    if err != nil {
        return fmt.Errorf("marshal activity: %w", err)
    }

    req, err := http.NewRequestWithContext(ctx, http.MethodPost, inboxURL, bytes.NewReader(body))
    if err != nil {
        return err
    }
    req.Header.Set("Content-Type", "application/activity+json")

    keyID := fmt.Sprintf("https://%s/ap/users/%s#main-key", s.cfg.Domain, signingHandle)
    if err := apub.SignRequest(req, keyID, s.crypto.privateKey); err != nil {
        return fmt.Errorf("sign request: %w", err)
    }

    resp, err := s.client.Do(req)
    if err != nil {
        return fmt.Errorf("deliver: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode &lt; 200 || resp.StatusCode &gt; 299 {
        return fmt.Errorf("delivery failed with status %d", resp.StatusCode)
    }

    return nil
}
```

```go
func (s *SpaceService) GetByID(ctx context.Context, id string) (*models.Space, error) {

    if err := s.cache.Get(ctx, "space:"+id, &amp;cached); err == nil {

        Handle:      row.Handle,
        DisplayName: row.DisplayName,
        PublicKey:   row.PublicKey,
        IsLocal:     row.IsLocal,
        CreatedAt:   row.CreatedAt,
    }

    return space, nil
}
```

```ts
import { useEffect, useRef, useCallback } from 'react'
import { useAuthStore } from '../store/auth'

type MessageHandler = (data: unknown) =&gt; void

export function useWebSocket(onMessage: MessageHandler) {
  const wsRef = useRef&lt;WebSocket | null&gt;(null)
  const token = useAuthStore((s) =&gt; s.accessToken)

  const connect = useCallback(() =&gt; {
    if (!token) return

    const proto = window.location.protocol === 'https:' ? 'wss' : 'ws'
    const ws = new WebSocket(
      `${proto}://${window.location.host}/ws`,
      // Pass token as subprotocol (standard WS auth pattern)
    )

    // Alternatively, append token as query param (simpler but less clean):
    // const ws = new WebSocket(`${proto}://${window.location.host}/ws?token=${token}`)

    ws.onmessage = (event) =&gt; {
      try {
        const data = JSON.parse(event.data)
        onMessage(data)
      } catch {
        // ignore malformed messages
      }
    }

    ws.onclose = () =&gt; {
      // Reconnect after 3 seconds
      setTimeout(connect, 3000)
    }

    wsRef.current = ws
  }, [token, onMessage])

  useEffect(() =&gt; {
    connect()
    return () =&gt; wsRef.current?.close()
  }, [connect])
}
```
