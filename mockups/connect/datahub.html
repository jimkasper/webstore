-- Mood on run days vs. non-run days (April 2026)
SELECT
  CASE WHEN r.actual_mins IS NOT NULL THEN 'Run day' ELSE 'Rest day' END AS day_type,
  AVG(j.day_rating)                  AS avg_day_rating,
  COUNT(*)                         AS days
FROM journal_entries j
LEFT JOIN activity_reflections r
  ON DATE(r.created_at) = j.entry_date
  AND r.event_id IN (SELECT id FROM events WHERE type_tag = 'exercise')
WHERE j.entry_date >= '2026-04-01'
GROUP BY day_type
ORDER BY day_type;
